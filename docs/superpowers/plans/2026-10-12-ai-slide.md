# AI활용 「슬라이드 디자인」 탭 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** AI활용 허브에 `/ai/slide/` 탭을 추가한다. 원천 슬라이드 갤러리와 같은 레이아웃·동작으로 만들고, 학교 주제로 새로 생성한 7개 예시 덱을 회원 전용 프롬프트와 함께 보여 준다.

**Architecture:** 모션 탭과 같은 3층 구조를 따른다.
- 공개 데이터: `content/slide/public.json`(git)
- 그림과 프롬프트: Railway Volume(`media/slide/…`, `private/slide-admin.json`)
- 화면: 서버 페이지가 manifest·권한을 읽고, 클라이언트 갤러리(필터·정렬·상세 창)에 props 로 넘긴다.

덱은 저장소 밖 원천 폴더(`/Volumes/Chois_SD2/dev/motion/slide/school/`)에서 NotebookLM 으로 만든다. `slide:import` → `media:push --set slide` 순서로 올린다.

**Tech Stack:** Next.js 16 App Router, React 19, Tailwind 4, zod 4, vitest 5, Playwright 1.64, NotebookLM CLI 0.8.4(`notebooklm`), poppler `pdftoppm`, `cwebp`.

**Spec:** `docs/superpowers/specs/2026-10-12-ai-slide-design.md`

## Global Constraints

- 원천 갤러리의 `app.js`·`slide-styles.css`·`hub.css`·문구(`name`·`look`·`use`)는 저장소에 복사하지 않는다. 코드와 문구는 새로 작성한다.
- 프롬프트는 git·빌드 산출물·공개 HTML 에 들어가면 안 된다. `content/slide/public.json` 은 `.strict()` 스키마로 검사한다.
- 탭 이름은 「슬라이드 디자인」, 주소는 `/ai/slide/`, 학교 이름은 「○○고등학교」, 1차 덱은 id `24` `59` `55` `04` `58` `10` `63` 의 7개다.
- 인기순 = 원천 배열 순서. 1차 7개의 순서는 `24, 59, 55, 04, 58, 10, 63` 이다.
- 그림 규격
  - 큰 그림 1376×768
  - 썸네일 줄 172×96
  - 목록 카드 760×424(표지·본문 장만)
  - 경로: `media/slide/<id>/{k}.webp`, `media/slide/<id>/s/{k}.webp`, `media/slide/<id>/t/{k}.webp`
- 반응형 규칙(`~/.claude/rules/responsive-ui.md`)
  - 라벨은 `whitespace-nowrap`, 칩 줄은 가로 스크롤
  - 터치 타겟은 `min-h-11`
  - 높이는 `dvh`, 모바일 바깥 여백은 `p-2` 이하
  - `break-words`·`break-all` 금지
- 주석은 "왜"가 비자명할 때만 쓴다. 사용자용 오류 문구는 한국어로 쓴다.

```ts
  { key: 'work', label: '업무·보고' },

  no: z.number().int().min(1),

  name: z.string().min(1),
  look: z.string().min(1),
  use: z.string().min(1),
  topic: z.string().min(1),
  slides: z.number().int().min(1).max(12),
  cover: z.number().int().min(1),
  body: z.number().int().min(1),

}).strict();

});
```

```ts
// slide 는 덱마다 장 수가 달라 파일 이름을 정해 둘 수 없으므로 media/slide 아래 webp 를 모두 훑는다.
export async function buildSlideManifest(root: string): Promise<MediaManifest> {
  const base = path.join(root, 'media/slide');
  if (!existsSync(base)) return {};
  const entries = await readdir(base, { recursive: true, withFileTypes: true });
  const out: MediaManifest = {};
  for (const d of entries.filter((e) => e.isFile() && e.name.endsWith('.webp'))) {
    const file = path.join(d.parentPath, d.name);
    const rel = path.relative(root, file).split(path.sep).join('/');
    const info = await stat(file);
    out[rel] = { size: info.size, sha256: await sha256File(file), updatedAt: info.mtime.toISOString() };
  }
  return mediaManifestSchema.parse(Object.fromEntries(Object.entries(out).sort(([a], [b]) => a.localeCompare(b))));
}
```

```ts
import { execFileSync } from 'node:child_process';
import { existsSync, mkdirSync, readFileSync, readdirSync, rmSync, writeFileSync } from 'node:fs';
import path from 'node:path';
import { pathToFileURL } from 'node:url';
import { slideAdminListSchema, slidePublicListSchema, type SlideAdmin, type SlidePublic } from '../content/slide/schema';
import { buildSlideManifest } from './media/manifest';

export type SchoolEntry = Omit<SlidePublic, 'slides'>;
export type DeckInfo = { pages: number; prompt: string };

export function buildSlideData(entries: SchoolEntry[], decks: Map<string, DeckInfo>): { publicList: SlidePublic[]; adminList: SlideAdmin[] } {
  const ready = entries.flatMap((e) => {
    const deck = decks.get(e.id);
    return deck ? [{ entry: e, deck }] : [];
  });
  return {
    publicList: slidePublicListSchema.parse(ready.map(({ entry, deck }) => ({ ...entry, slides: deck.pages }))),
    adminList: slideAdminListSchema.parse(ready.map(({ entry, deck }) => ({ id: entry.id, prompt: deck.prompt.trim() }))),
  };
}

const run = (cmd: string, args: string[]) => execFileSync(cmd, args, { stdio: ['ignore', 'ignore', 'inherit'] });

function renderDeck(pdf: string, dir: string, tmp: string, cards: number[]): number {
  rmSync(dir, { recursive: true, force: true });
  rmSync(tmp, { recursive: true, force: true });
  for (const d of [dir, path.join(dir, 's'), path.join(dir, 't'), tmp]) mkdirSync(d, { recursive: true });
  run('pdftoppm', ['-png', '-scale-to-x', '1376', '-scale-to-y', '-1', pdf, path.join(tmp, 'p')]);
  const pngs = readdirSync(tmp)
    .filter((f) => /^p-\d+\.png$/.test(f))
    .sort((a, b) => Number(a.match(/\d+/)?.[0]) - Number(b.match(/\d+/)?.[0]));
  pngs.forEach((png, i) => {
    const k = i + 1;
    const src = path.join(tmp, png);
    run('cwebp', ['-quiet', '-q', '82', '-resize', '1376', '768', src, '-o', path.join(dir, `${k}.webp`)]);
    run('cwebp', ['-quiet', '-q', '70', '-resize', '172', '96', src, '-o', path.join(dir, 's', `${k}.webp`)]);
    if (cards.includes(k)) run('cwebp', ['-quiet', '-q', '78', '-resize', '760', '424', src, '-o', path.join(dir, 't', `${k}.webp`)]);
  });
  rmSync(tmp, { recursive: true, force: true });
  return pngs.length;
}

async function main() {
  const src = process.env.SLIDE_SRC ?? '/Volumes/Chois_SD2/dev/motion/slide/school';
  const out = path.resolve('.media-out');
  const entries = JSON.parse(readFileSync(path.join(src, 'manifest.json'), 'utf8')) as SchoolEntry[];
  const decks = new Map<string, DeckInfo>();
  for (const e of entries) {
    const pdf = path.join(src, e.id, 'deck.pdf');
    if (!existsSync(pdf)) {
      console.warn(`skip ${e.id}: deck.pdf missing`);
      continue;
    }
    const pages = renderDeck(pdf, path.join(out, 'media/slide', e.id), path.join(out, '.tmp-slide'), [e.cover, e.body]);
    decks.set(e.id, { pages, prompt: readFileSync(path.join(src, e.id, 'prompt.txt'), 'utf8') });
    console.info(`rendered ${e.id}: ${pages} pages`);
  }
  const { publicList, adminList } = buildSlideData(entries, decks);
  writeFileSync(path.resolve('content/slide/public.json'), JSON.stringify(publicList, null, 1) + '\n');
  mkdirSync(path.join(out, 'private'), { recursive: true });
  writeFileSync(path.join(out, 'private/slide-admin.json'), JSON.stringify(adminList, null, 1) + '\n');
  const manifest = await buildSlideManifest(out);
  writeFileSync(path.join(out, 'media/slide/manifest.json'), JSON.stringify(manifest, null, 1) + '\n');
  console.info(`public ${publicList.length} · manifest ${Object.keys(manifest).length} files`);
}
```

```ts
import type { SlideFamily } from '@me/content';

export type SortKey = 'pop' | 'no' | 'new';
export type ViewKey = 'cover' | 'body';
export type GalleryQuery = { family: SlideFamily | null; sort: SortKey; view: ViewKey };
export type Orderable = { id: string; no: number; family: SlideFamily; added: string };
export type ViewerPos = { style: number; slide: number };

const SORTS: readonly string[] = ['pop', 'no', 'new'];

export function parseQuery(params: URLSearchParams, present: readonly string[]): GalleryQuery {
  const f = params.get('f');
  const s = params.get('s') ?? '';
  return {
    family: f && present.includes(f) ? (f as SlideFamily) : null,
    sort: SORTS.includes(s) ? (s as SortKey) : 'pop',
    view: params.get('v') === 'body' ? 'body' : 'cover',
  };
}

export function toSearch(q: GalleryQuery): string {
  const p = new URLSearchParams();
  if (q.family) p.set('f', q.family);
  if (q.sort !== 'pop') p.set('s', q.sort);
  if (q.view !== 'cover') p.set('v', q.view);
  const s = p.toString();
  return s ? `?${s}` : '';
}

// pop 은 원천 배열 순서 그대로다. 같은 날 올린 스타일은 번호가 큰 쪽을 최신으로 본다.
export function arrange<T extends Orderable>(items: readonly T[], q: GalleryQuery): T[] {
  const shown = q.family ? items.filter((i) => i.family === q.family) : [...items];
  if (q.sort === 'no') shown.sort((a, b) => a.no - b.no);
  if (q.sort === 'new') shown.sort((a, b) => b.added.localeCompare(a.added) || b.no - a.no);
  return shown;
}

// #번호로 연 스타일이 지금 거르기에 없으면 거르기를 무시한 순서로 넘긴다.
export function pickOrder<T extends Orderable>(items: readonly T[], q: GalleryQuery, id: string): T[] {
  const shown = arrange(items, q);
  return shown.some((i) => i.id === id) ? shown : arrange(items, { ...q, family: null });
}

export function stepSlide(pos: ViewerPos, dir: 1 | -1, counts: readonly number[]): ViewerPos {
  const next = pos.slide + dir;
  if (next >= 0 && next < (counts[pos.style] ?? 1)) return { style: pos.style, slide: next };
  const style = (pos.style + dir + counts.length) % counts.length;
  return { style, slide: dir === 1 ? 0 : (counts[style] ?? 1) - 1 };
}

export function stepStyle(pos: ViewerPos, dir: 1 | -1, n: number): ViewerPos {
  return { style: (pos.style + dir + n) % n, slide: 0 };
}
```

```ts
import { useSyncExternalStore } from 'react';

const EVENT = 'slide-hash-change';

export function readHashId(hash: string): string {
  try {
    return decodeURIComponent(hash.replace(/^#/, ''));
  } catch {
    return '';
  }
}

function subscribe(onChange: () => void) {
  window.addEventListener('hashchange', onChange);
  window.addEventListener(EVENT, onChange);
  return () => {
    window.removeEventListener('hashchange', onChange);
    window.removeEventListener(EVENT, onChange);
  };
}

// 해시가 상세 창의 유일한 진실 공급원이다. 서버 렌더에는 해시가 없으므로 항상 닫힌 상태로 그린다.
export function useHashId(): string {
  return useSyncExternalStore(subscribe, () => readHashId(window.location.hash), () => '');
}

// 상세 창 안에서 스타일을 넘길 때 뒤로 가기 기록이 쌓이지 않게 replaceState 를 쓴다. replaceState 는 hashchange 를 내지 않아 직접 알린다.
export function setHashId(id: string | null) {
  history.replaceState(null, '', id ? `#${encodeURIComponent(id)}` : `${location.pathname}${location.search}`);
  window.dispatchEvent(new Event(EVENT));
}
```

```tsx
'use client';
import { useEffect, useRef, useState } from 'react';
import type { SortKey } from './slide-order';

const LABEL: Record<SortKey, string> = { pop: '인기순', no: '번호순', new: '최신순' };

        type="button"

        <div role="menu" className="absolute right-0 z-10 mt-1 flex min-w-32 flex-col rounded-xl border border-line bg-bg p-1 shadow-lg">
```

```tsx
'use client';
import { useSearchParams } from 'next/navigation';
import { useState } from 'react';
import type { SlideFamily } from '@me/content';
import type { SlidePromptAccess } from '@/lib/access';
import type { SlideCardData } from '@/lib/slides';
import { SlideCard } from './SlideCard';
import { SlideSortMenu } from './SlideSortMenu';
import { SlideViewer } from './SlideViewer';
import { setHashId, useHashId } from './slide-hash';
import { arrange, parseQuery, pickOrder, toSearch, type GalleryQuery, type ViewerPos } from './slide-order';

type Family = { readonly key: SlideFamily; readonly label: string };
type Props = { items: SlideCardData[]; families: readonly Family[]; access: SlidePromptAccess };

const chip = 'min-h-11 shrink-0 rounded-full border px-4 text-sm whitespace-nowrap';
const VIEWS = [{ key: 'cover', label: '표지' }, { key: 'body', label: '본문' }] as const;

export function SlideGallery({ items, families, access }: Props) {
  const present = families.filter((f) => items.some((i) => i.family === f.key));
  // URL 이 거르기·정렬·보기의 유일한 진실 공급원이다(새로고침·공유해도 같은 화면).
  const query = parseQuery(useSearchParams(), present.map((f) => f.key));
  const shown = arrange(items, query);
  const hashId = useHashId();
  const [slideOf, setSlideOf] = useState<{ id: string; k: number }>({ id: '', k: 0 });

  const order = pickOrder(items, query, hashId);
  const styleIndex = order.findIndex((s) => s.id === hashId);
  const pos: ViewerPos | null = styleIndex < 0 ? null : { style: styleIndex, slide: slideOf.id === hashId ? slideOf.k : 0 };

  const update = (patch: Partial<GalleryQuery>) => {
    history.replaceState(null, '', `${location.pathname}${toSearch({ ...query, ...patch })}`);
  };
  const open = (id: string, k = 0) => {
    setSlideOf({ id, k });
    setHashId(id);
  };
  const labelOf = (key: SlideFamily) => families.find((f) => f.key === key)?.label ?? key;
  const countOf = (key: SlideFamily | null) => (key ? items.filter((i) => i.family === key).length : items.length);

  return (
    <div data-testid="slide-gallery">
      <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
        <div role="toolbar" aria-label="분류로 거르기" data-testid="slide-chips" className="-mx-2 flex min-w-0 flex-1 gap-2 overflow-x-auto px-2 pb-1">
          {[null, ...present].map((f) => {
            const key = f?.key ?? null;
            const isOn = query.family === key;
            return (
              <button key={key ?? 'all'} type="button" aria-pressed={isOn} onClick={() => update({ family: key })} className={`${chip} ${isOn ? 'border-fg bg-fg text-bg' : 'border-line'}`}>
                {f?.label ?? '전체'} <span className="font-mono text-xs opacity-70">{countOf(key)}</span>
              </button>
            );
          })}
        </div>
        <div className="flex shrink-0 items-center gap-2">
          <SlideSortMenu value={query.sort} onChange={(sort) => update({ sort })} />
          <div role="group" aria-label="목록에 보여 줄 장" data-testid="slide-view" className="flex rounded-full border border-line p-0.5">
            {VIEWS.map((v) => (
              <button key={v.key} type="button" aria-pressed={query.view === v.key} onClick={() => update({ view: v.key })} className={`min-h-11 rounded-full px-4 text-sm whitespace-nowrap ${query.view === v.key ? 'bg-fg text-bg' : ''}`}>
                {v.label}
              </button>
            ))}
          </div>
        </div>
      </div>
      <p role="status" className="sr-only">{`${shown.length}개 스타일`}</p>
      <ol className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
        {shown.map((s) => (
          <li key={s.id}>
            <SlideCard item={s} view={query.view} familyLabel={labelOf(s.family)} onOpen={() => open(s.id)} />
          </li>
        ))}
      </ol>
      {pos ? (
        <SlideViewer
          list={order}
          pos={pos}
          onMove={(next) => {
            const id = order[next.style]?.id;
            if (id) open(id, next.slide);
          }}
          onClose={() => setHashId(null)}
          familyLabel={labelOf}
          access={access}
        />
      ) : null}
    </div>
  );
}
```

```ts
export async function writeClipboard(text: string) {
  try {
    await navigator.clipboard.writeText(text);
  } catch {
    const ta = document.createElement('textarea');
    ta.value = text;
    document.body.appendChild(ta);
    ta.select();
    document.execCommand('copy');
    ta.remove();
  }
}
```

```tsx
'use client';
import { useEffect, useRef } from 'react';
import type { SlideFamily } from '@me/content';
import type { SlidePromptAccess } from '@/lib/access';
import type { SlideCardData } from '@/lib/slides';
import { SlidePrompt } from './SlidePrompt';
import { stepSlide, stepStyle, type ViewerPos } from './slide-order';

type Props = {
  list: SlideCardData[];
  pos: ViewerPos;
  onMove: (pos: ViewerPos) => void;
  onClose: () => void;
  familyLabel: (key: SlideFamily) => string;
  access: SlidePromptAccess;
};

const SWIPE_PX = 40;
const iconBtn = 'inline-flex min-h-11 min-w-11 items-center justify-center rounded-full text-lg hover:bg-white/10';
const navBtn = 'absolute top-1/2 inline-flex min-h-11 min-w-11 -translate-y-1/2 items-center justify-center rounded-full bg-black/50 text-2xl text-white';

export function SlideViewer({ list, pos, onMove, onClose, familyLabel, access }: Props) {
  const dialog = useRef<HTMLDialogElement>(null);
  const stage = useRef<HTMLDivElement>(null);
  const swipeStart = useRef<number | null>(null);
  const counts = list.map((s) => s.slides);

  useEffect(() => {
    const d = dialog.current;
    if (d && !d.open) d.showModal();
  }, []);

  // 의존성 배열 없이 매 렌더 다시 건다 — 최신 pos 로 넘겨야 한다.
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.altKey || e.ctrlKey || e.metaKey) return;
      if (e.key === 'ArrowRight') onMove(stepSlide(pos, 1, counts));
      if (e.key === 'ArrowLeft') onMove(stepSlide(pos, -1, counts));
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  });

  const style = list[pos.style];
  if (!style) return null;
  const go = (dir: 1 | -1) => onMove(stepSlide(pos, dir, counts));
  const goStyle = (dir: 1 | -1) => onMove(stepStyle(pos, dir, list.length));
  const toggleFull = () => {
    if (document.fullscreenElement) void document.exitFullscreen();
    else void stage.current?.requestFullscreen();
  };

  return (
    <dialog
      ref={dialog}
      data-testid="slide-viewer"
      aria-labelledby="slide-viewer-name"
      onClose={onClose}
      className="m-0 h-dvh max-h-none w-screen max-w-none bg-dark-bg p-0 text-dark-fg backdrop:bg-black/80"
    >
      <div className="flex h-full flex-col overflow-y-auto lg:flex-row lg:overflow-hidden">
        <div className="flex min-h-0 shrink-0 flex-col gap-2 p-2 sm:p-3 lg:flex-1 lg:p-4">
          <div
            ref={stage}
            data-testid="slide-stage"
            className="relative flex aspect-video w-full touch-pan-y items-center justify-center bg-black lg:aspect-auto lg:min-h-0 lg:flex-1"
            onPointerDown={(e) => { swipeStart.current = e.clientX; }}
            onPointerUp={(e) => {
              const start = swipeStart.current;
              swipeStart.current = null;
              if (start !== null && Math.abs(e.clientX - start) >= SWIPE_PX) go(e.clientX < start ? 1 : -1);
            }}
          >
            {/* eslint-disable-next-line @next/next/no-img-element -- Volume 에서 스트리밍하는 버전 URL 이라 next/image 대상이 아니다 */}
            <img data-testid="slide-viewer-img" src={style.images[pos.slide]} alt={`${style.name} ${pos.slide + 1}번째 장`} width={1376} height={768} draggable={false} className="max-h-full max-w-full object-contain select-none" />
            <button type="button" aria-label="이전 슬라이드" title="이전 슬라이드 (←)" onClick={() => go(-1)} className={`${navBtn} left-1`}>‹</button>
            <button type="button" aria-label="다음 슬라이드" title="다음 슬라이드 (→)" onClick={() => go(1)} className={`${navBtn} right-1`}>›</button>
            <button type="button" aria-label="전체 화면 전환" title="전체 화면" onClick={toggleFull} className={`${iconBtn} absolute top-1 right-1 bg-black/50 text-white`}>⛶</button>
          </div>
          <div className="flex items-center gap-2">
            <div role="group" aria-label="이 스타일의 예시 슬라이드" className="flex min-w-0 flex-1 gap-1.5 overflow-x-auto pb-1">
              {style.thumbs.map((src, k) => (
                <button
                  key={src}
                  type="button"
                  aria-label={`${k + 1}번째 장`}
                  aria-current={k === pos.slide ? 'true' : undefined}
                  onClick={() => onMove({ style: pos.style, slide: k })}
                  className={`h-12 shrink-0 overflow-hidden rounded border-2 sm:h-14 ${k === pos.slide ? 'border-accent-soft' : 'border-transparent opacity-60'}`}
                >
                  {/* eslint-disable-next-line @next/next/no-img-element -- 위와 같음 */}
                  <img src={src} alt="" width={172} height={96} className="h-full w-auto" />
                </button>
              ))}
            </div>
            <span data-testid="slide-pos" className="shrink-0 font-mono text-xs whitespace-nowrap">{pos.slide + 1} / {style.slides}</span>
          </div>
        </div>
        <aside className="flex flex-col gap-3 border-t border-white/10 p-3 lg:w-96 lg:shrink-0 lg:overflow-y-auto lg:border-t-0 lg:border-l lg:p-5">
```

```ts
export const SLIDE_RENDERED = ['24', '63'];
const slideList = JSON.parse(readFileSync(path.resolve(import.meta.dirname, '../content/slide/public.json'), 'utf8')) as { id: string; slides: number; cover: number; body: number }[];
```

```ts
import { readFileSync } from 'node:fs';
import path from 'node:path';
import { expect, test } from '@playwright/test';
import { arrange } from '../src/components/slide-gallery/slide-order';
import { MARKER, signInAs } from './session';

const PAGE = '/ai/slide/';
type Item = { id: string; no: number; name: string; family: 'hand' | 'work'; added: string; slides: number; cover: number; body: number };
const all = JSON.parse(readFileSync(path.resolve('../content/slide/public.json'), 'utf8')) as Item[];
const rendered = all.filter((s) => ['24', '63'].includes(s.id));
const cards = (page: import('@playwright/test').Page) => page.getByTestId(/^slide-card-\d+$/);

test('hub tab shows and only rendered styles are listed', async ({ page }) => {
  await page.goto(PAGE);
  await expect(page.getByRole('link', { name: '슬라이드 디자인' })).toHaveAttribute('aria-current', 'page');
  await expect(cards(page)).toHaveCount(rendered.length);
});

test('family chip filters and survives reload', async ({ page }) => {
  await page.goto(PAGE);
  await page.getByTestId('slide-chips').getByRole('button', { name: /업무·보고/ }).click();
  await expect(cards(page)).toHaveCount(1);
  await expect(page).toHaveURL(/\?f=work$/);
  await page.reload();
  await expect(cards(page)).toHaveCount(1);
  await page.getByTestId('slide-chips').getByRole('button', { name: /전체/ }).click();
  await expect(cards(page)).toHaveCount(rendered.length);
});

test('sort menu reorders cards', async ({ page }) => {
  await page.goto(PAGE);
  for (const [label, sort] of [['번호순', 'no'], ['최신순', 'new'], ['인기순', 'pop']] as const) {
    await page.getByTestId('slide-sort').getByRole('button', { name: /정렬/ }).click();
    await page.getByRole('menuitemradio', { name: label }).click();
    const want = arrange(rendered, { family: null, sort, view: 'cover' }).map((s) => `slide-card-${s.id}`);
    await expect(cards(page)).toHaveCount(want.length);
    expect(await cards(page).evaluateAll((els) => els.map((e) => e.getAttribute('data-testid')))).toEqual(want);
  }
});

test('cover/body toggle swaps the card image', async ({ page }) => {
  const s = rendered[0]!;
  await page.goto(PAGE);
  const front = page.getByTestId(`slide-card-${s.id}`).getByTestId('slide-card-front');
  await expect(front).toHaveAttribute('src', new RegExp(`/media/slide/${s.id}/t/${s.cover}\\.webp`));
  await page.getByTestId('slide-view').getByRole('button', { name: '본문' }).click();
  await expect(front).toHaveAttribute('src', new RegExp(`/media/slide/${s.id}/t/${s.body}\\.webp`));
  await expect(page).toHaveURL(/v=body/);
});

test('viewer pages through slides into the next style and closes with Escape', async ({ page }) => {
  const [first, second] = rendered as [Item, Item];
  await page.goto(PAGE);
  await page.getByTestId(`slide-card-${first.id}`).click();
  const viewer = page.getByTestId('slide-viewer');
  await expect(viewer).toBeVisible();
  await expect(page).toHaveURL(new RegExp(`#${first.id}$`));
  await expect(viewer.getByTestId('slide-pos')).toHaveText(`1 / ${first.slides}`);
  for (let i = 0; i < first.slides; i++) await page.keyboard.press('ArrowRight');
  await expect(viewer.getByRole('heading', { name: second.name })).toBeVisible();
  await expect(viewer.getByTestId('slide-pos')).toHaveText(`1 / ${second.slides}`);
  await page.keyboard.press('ArrowLeft');
  await expect(viewer.getByRole('heading', { name: first.name })).toBeVisible();
  await expect(viewer.getByTestId('slide-pos')).toHaveText(`${first.slides} / ${first.slides}`);
  await page.keyboard.press('Escape');
  await expect(viewer).toBeHidden();
  await expect(page).toHaveURL(/\/ai\/slide\/$/);
});

test('hash deep links open the style, even when filtered out, and bad hashes are ignored', async ({ page }) => {
  const target = rendered.find((s) => s.family === 'work')!;
  await page.goto(`${PAGE}?f=hand#${target.id}`);
  await expect(page.getByTestId('slide-viewer').getByRole('heading', { name: target.name })).toBeVisible();
  for (const bad of ['#999', '#%E0']) {
    await page.goto(`${PAGE}${bad}`);
    await expect(cards(page).first()).toBeVisible();
    await expect(page.getByTestId('slide-viewer')).toHaveCount(0);
  }
});

test('anonymous copy opens the sign-in dialog and html has no prompt', async ({ page }) => {
  expect(await (await page.request.get(PAGE)).text()).not.toContain(MARKER);
  await page.goto(`${PAGE}#${rendered[0]!.id}`);
  await expect(page.getByTestId('slide-prompt-locked')).toBeVisible();
  await page.getByTestId('copy-slide-prompt').click();
  await expect(page.getByTestId('signin-dialog')).toBeVisible();
});

test('approved member sees and copies the prompt', async ({ browser }) => {
  test.skip(!process.env.E2E_DATABASE_URL, 'DB 필요');
  const ctx = await browser.newContext({ permissions: ['clipboard-read', 'clipboard-write'] });
  await signInAs(ctx, 'viewer@example.com');
  const page = await ctx.newPage();

  await page.getByTestId('copy-slide-prompt').click();
  await expect(page.getByTestId('copy-slide-prompt')).toHaveText('복사됨');

});

test('mobile layout has no horizontal page scroll', async ({ page }) => {
  await page.setViewportSize({ width: 375, height: 812 });
  await page.goto(PAGE);
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= window.innerWidth)).toBe(true);
```
